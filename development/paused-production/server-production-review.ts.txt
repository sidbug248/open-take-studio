import {Router,type Request} from 'express';
import {createHash,randomBytes,randomUUID,timingSafeEqual} from 'node:crypto';
import {z} from 'zod';
import type {ReviewBoard,PublicReviewBoard} from '../shared/production-types.ts';
import type {Store} from './store.ts';
import type {LocalFileStorage} from './media.ts';
import {bad,localTake,projectAsset} from './production-core.ts';

export const publicReview=({tokenHash,...review}:ReviewBoard):PublicReviewBoard=>review;
const tokenHash=(token:string)=>createHash('sha256').update(token).digest('hex');
export function authorizeReview(store:Store,reviewId:string,token:string){const row=store.getScopedRecord<ReviewBoard>('review',reviewId),hash=tokenHash(token);if(!row||row.data.revoked||Date.parse(row.data.expiresAt)<=Date.now()||!row.data.tokenHash||!timingSafeEqual(Buffer.from(hash),Buffer.from(row.data.tokenHash)))throw bad('This review link is unavailable or has expired.',404);return row;}
async function auditReview(store:Store,media:LocalFileStorage,review:ReviewBoard){const safe=publicReview(review),asset=await media.put({projectId:review.projectId,name:review.name+' · review snapshot.json',mimeType:'application/json',kind:'document',bytes:Buffer.from(JSON.stringify(safe,null,2)),provenance:'local-edit',metadata:{operation:'review-snapshot',reviewDocument:safe,inputAssetIds:{sources:review.assetIds}}});const take=localTake(review.projectId,asset,'review-snapshot',review.assetIds,review.name+' · review snapshot');store.transaction(()=>{store.upsertAsset(asset);store.upsertTake(take);});return{asset,take};}
const save=(store:Store,review:ReviewBoard)=>store.upsertScopedRecord({id:review.id,projectId:review.projectId,kind:'review',createdAt:review.createdAt,updatedAt:review.updatedAt,data:review});
export function reviewOwnerRoutes(store:Store,media:LocalFileStorage){const api=Router();
 api.post('/projects/:id/production/reviews',async(req,res)=>{const body=z.object({name:z.string().min(1).max(160),assetIds:z.array(z.string()).min(1).max(30),allowComments:z.boolean().default(true),expiresInDays:z.number().int().min(1).max(90).default(7)}).strict().parse(req.body);for(const id of body.assetIds)projectAsset(store,req.params.id,id);const now=new Date().toISOString(),token=randomBytes(32).toString('base64url'),review:ReviewBoard={id:randomUUID(),projectId:req.params.id,name:body.name,assetIds:[...new Set(body.assetIds)],allowComments:body.allowComments,createdAt:now,updatedAt:now,expiresAt:new Date(Date.now()+body.expiresInDays*86400000).toISOString(),revoked:false,tokenHash:tokenHash(token),comments:[]};save(store,review);await auditReview(store,media,review);res.status(201).json({review:publicReview(review),url:`/review/${review.id}/${token}`});});
 api.patch('/projects/:id/production/reviews/:reviewId',async(req,res)=>{const body=z.object({revoke:z.boolean().optional(),commentId:z.string().optional(),resolved:z.boolean().optional()}).strict().parse(req.body),row=store.getScopedRecord<ReviewBoard>('review',req.params.reviewId);if(!row||row.projectId!==req.params.id)throw bad('Review not found.',404);const data={...row.data,comments:row.data.comments.map(c=>({...c})),updatedAt:new Date().toISOString()};if(body.revoke)data.revoked=true;if(body.commentId){const comment=data.comments.find(c=>c.id===body.commentId);if(!comment)throw bad('Comment not found.',404);comment.resolved=body.resolved??true;}save(store,data);await auditReview(store,media,data);res.json(publicReview(data));});
 return api;
}

/** Mounted outside the owner API gate. Every route checks its own expiring share token. */
export function createReviewRouter(store:Store,media:LocalFileStorage){const api=Router(),limits=new Map<string,{at:number;count:number}>();api.use((_req,res,next)=>{res.set({'Cache-Control':'private, no-store','Referrer-Policy':'no-referrer','X-Robots-Tag':'noindex, nofollow','X-Content-Type-Options':'nosniff'});next();});
 api.get('/review/:reviewId/:token',(req,res)=>{const row=authorizeReview(store,req.params.reviewId,req.params.token);res.type('html').send(reviewHtml(row.data,store,`/review/${row.id}/${req.params.token}`));});
 api.get('/review/:reviewId/:token/media/:assetId',(req,res)=>{const row=authorizeReview(store,req.params.reviewId,req.params.token);if(!row.data.assetIds.includes(req.params.assetId))throw bad('This asset is not in the review.',404);const asset=projectAsset(store,row.projectId,req.params.assetId);if(req.query.download)res.attachment(asset.name);res.type(asset.mimeType).sendFile(media.path(asset));});
 api.post('/review/:reviewId/:token/comments',async(req,res)=>{const row=authorizeReview(store,req.params.reviewId,req.params.token);if(!row.data.allowComments)throw bad('Comments are disabled on this review.',403);const now=Date.now(),rate=limits.get(row.id);if(rate&&now-rate.at<60000&&rate.count>=20)throw bad('Please wait a moment before adding another comment.',429);limits.set(row.id,{at:rate&&now-rate.at<60000?rate.at:now,count:rate&&now-rate.at<60000?rate.count+1:1});if(limits.size>2000)for(const [id,value]of limits)if(now-value.at>60000)limits.delete(id);
  const body=z.object({assetId:z.string(),author:z.string().trim().min(1).max(80),body:z.string().trim().min(1).max(4000),seconds:z.number().min(0).max(86400).optional(),decision:z.enum(['approve','changes']).optional(),replyTo:z.string().optional()}).strict().parse(req.body);if(!row.data.assetIds.includes(body.assetId))throw bad('This asset is not in the review.',404);if(row.data.comments.length>=1000)throw bad('This review has reached its comment limit.',409);if(body.replyTo&&!row.data.comments.some(c=>c.id===body.replyTo&&c.assetId===body.assetId))throw bad('Reply to a comment on this same asset.');const data={...row.data,updatedAt:new Date().toISOString(),comments:[...row.data.comments,{...body,id:randomUUID(),createdAt:new Date().toISOString(),resolved:false}]};save(store,data);await auditReview(store,media,data);res.status(201).json({comment:data.comments.at(-1)});
 });return api;}
