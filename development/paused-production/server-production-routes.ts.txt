import {Router} from 'express';
import {createHash,randomUUID} from 'node:crypto';
import {z} from 'zod';
import type {Store} from './store.ts';
import type {LocalFileStorage} from './media.ts';
import type {AppliedStory,DeliverySchedule,GraphRun,ProductionGraph,ProductionSnapshot,StoryDocument} from '../shared/production-types.ts';
import {bad,deliverySpecSchema,fingerprint,graphGenerationRequest,graphOrder,parseStory,prepareDocument,productionSchema,projectAsset,readDocument,renderStory} from './production-core.ts';
import {createSchedule,prepareDelivery} from './production-delivery.ts';
import {publicReview,reviewOwnerRoutes} from './production-review.ts';

const intentSchema=z.string().min(8).max(100);
function saveRun(store:Store,run:GraphRun){run.updatedAt=new Date().toISOString();store.upsertScopedRecord({id:run.id,projectId:run.projectId,kind:'production-run',createdAt:run.createdAt,updatedAt:run.updatedAt,data:run});return run;}
export async function advanceGraph(store:Store,media:LocalFileStorage,run:GraphRun){
 const graph=readDocument(store,run.projectId,run.graphAssetId) as ProductionGraph;if(graph.kind!=='graph')throw bad('Choose a production canvas.');
 for(const id of graphOrder(graph)){const node=graph.nodes.find(n=>n.id===id)!,state=run.nodes.find(n=>n.nodeId===id)!;if(state.jobId){const job=store.getJob(state.jobId),take=job&&store.getTake(job.takeId);if(job?.state==='COMPLETED'&&take?.assetIds.length){state.state='complete';state.assetIds=take.assetIds;state.takeId=take.id;}else if(job&&['FAILED','BLOCKED','EXPIRED'].includes(job.state)){state.state='failed';state.error=job.errorMessage||job.state;}else state.state='waiting';continue;}if(state.state==='complete')continue;
  const incoming=graph.edges.filter(e=>e.to===id).map(e=>run.nodes.find(s=>s.nodeId===e.from)!);if(incoming.some(s=>s.state!=='complete')){state.state='waiting';continue;}const sources=[...new Set(incoming.flatMap(s=>s.assetIds))];
  try{if(node.kind==='asset'){state.assetIds=[projectAsset(store,run.projectId,node.assetId).id];state.state='complete';}
   else if(node.kind==='recipe'){state.request??=graphGenerationRequest(store,run.projectId,graph,run,id);state.state='ready';}
   else if(node.kind==='storyboard'){if(!sources.length)throw bad('Connect at least one image to this storyboard.');const key='graph-local:'+run.id+':'+id,existing=store.getOperationReceipt<{assetId:string;takeId:string}>(key);if(existing){state.assetIds=[existing.result.assetId];state.takeId=existing.result.takeId;}else{const result=await renderStory(store,media,run.projectId,node.label,sources.map(assetId=>{const a=projectAsset(store,run.projectId,assetId);return{title:a.name,direction:'',assetId};}),node.columns);store.transaction(()=>{store.upsertAsset(result.asset);store.upsertTake(result.take);store.putOperationReceipt({key,fingerprint:fingerprint(sources),result:{assetId:result.asset.id,takeId:result.take.id},createdAt:new Date().toISOString()});});state.assetIds=[result.asset.id];state.takeId=result.take.id;}state.state='complete';}
   else {if(!sources.length)throw bad('Connect media to this delivery package.');const result=await prepareDelivery(store,media,run.projectId,{name:node.label,assetIds:sources,preset:node.preset,caption:node.caption},'graph-'+run.id+'-'+id);state.assetIds=[result.asset.id];state.takeId=result.take.id;state.state='complete';}
   state.error=undefined;
  }catch(error){state.state='failed';state.error=error instanceof Error?error.message:'Node failed.';}
  saveRun(store,run);
 }
 return saveRun(store,run);
}
export async function applyStory(store:Store,media:LocalFileStorage,projectId:string,raw:unknown,clientRequestId:string,parentAssetId?:string):Promise<AppliedStory>{
 const document=productionSchema.parse(raw);if(document.kind!=='story')throw bad('Choose a story plan.');const key='story-plan:'+projectId+':'+clientRequestId,hash=fingerprint({document,parentAssetId});const prior=store.getOperationReceipt<AppliedStory>(key);if(prior){if(prior.fingerprint!==hash)throw bad('This story identity belongs to a different revision.',409);return prior.result;}
 const position=store.listShots(projectId).reduce((n,s)=>Math.max(n,s.position+1),0);const shots=document.beats.map((b,index)=>({id:b.shotId||randomUUID(),projectId,name:b.title,description:b.direction+(b.narration?'\n\nNarration: '+b.narration:''),position:store.getShot(b.shotId||'')?.position??position+index,selectedTakeId:b.assetId?store.listTakes(projectId).find(t=>t.assetIds.includes(b.assetId!))?.id:undefined}));
 const updated:StoryDocument={...document,beats:document.beats.map((b,i)=>({...b,shotId:shots[i].id}))};
 // Validate references before planning shots; newly assigned shot ids are committed below.
 const prepared=await prepareDocument(store,media,projectId,{...updated,beats:updated.beats.map(({shotId,...b})=>b)},parentAssetId);prepared.document=updated;prepared.asset.metadata.productionDocument=updated;
 const page=await renderStory(store,media,projectId,document.name,document.beats.map(b=>({title:b.title,direction:b.direction,assetId:b.assetId})));page.asset.metadata.sourceAssetId=prepared.asset.id;page.take.sourceAssetId=prepared.asset.id;page.take.inputAssetIds.document=prepared.asset.id;page.take.parentTakeId=prepared.take.id;const result={...prepared,page,shots};
 return store.transaction(()=>{const raced=store.getOperationReceipt<AppliedStory>(key);if(raced){if(raced.fingerprint!==hash)throw bad('Story identity conflict.',409);return raced.result;}for(const shot of shots)store.upsertShot(shot);store.upsertAsset(prepared.asset);store.upsertTake(prepared.take);store.upsertAsset(page.asset);store.upsertTake(page.take);store.putOperationReceipt({key,fingerprint:hash,result,createdAt:new Date().toISOString()});return result;});
}
export function createProductionRouter(store:Store,media:LocalFileStorage){
 const api=Router();
 api.get('/projects/:id/production',(req,res)=>{const projectId=req.params.id;if(!store.getProject(projectId))throw bad('Project not found.',404);const value:ProductionSnapshot={documents:store.listAssets(projectId).filter(a=>!!a.metadata.productionDocument),deliveries:store.listScopedRecords<DeliverySchedule>('delivery',projectId).map(r=>r.data),reviews:store.listScopedRecords<import('../shared/production-types.ts').ReviewBoard>('review',projectId).map(r=>publicReview(r.data)),runs:store.listScopedRecords<GraphRun>('production-run',projectId).map(r=>r.data)};res.json(value);});
 api.post('/projects/:id/production/story/parse',async(req,res)=>{const body=z.object({text:z.string().max(100000).optional(),sourceAssetId:z.string().optional(),name:z.string().max(160).default('Untitled treatment')}).parse(req.body);let text=body.text??'';if(body.sourceAssetId){const a=projectAsset(store,req.params.id,body.sourceAssetId);if(!['text/plain','text/markdown'].includes(a.mimeType)||a.size>512000)throw bad('Use a plain text or Markdown document, up to 500 KB.');text=(await media.read(a)).toString('utf8');}res.json({...parseStory(text,body.name),sourceAssetId:body.sourceAssetId});});
 api.post('/projects/:id/production/documents',async(req,res)=>{const body=z.object({document:productionSchema,parentAssetId:z.string().optional(),clientRequestId:intentSchema}).parse(req.body),key='production-save:'+req.params.id+':'+body.clientRequestId,hash=fingerprint(body),old=store.getOperationReceipt(key);if(old){if(old.fingerprint!==hash)throw bad('This save identity belongs to another revision.',409);res.json(old.result);return;}const result=await prepareDocument(store,media,req.params.id,body.document,body.parentAssetId);store.transaction(()=>{store.upsertAsset(result.asset);store.upsertTake(result.take);store.putOperationReceipt({key,fingerprint:hash,result,createdAt:new Date().toISOString()});});res.status(201).json(result);});
 api.post('/projects/:id/production/story/apply',async(req,res)=>{const body=z.object({document:productionSchema,parentAssetId:z.string().optional(),clientRequestId:intentSchema}).parse(req.body);res.status(201).json(await applyStory(store,media,req.params.id,body.document,body.clientRequestId,body.parentAssetId));});
 api.post('/projects/:id/production/graphs/run',async(req,res)=>{const body=z.object({graphAssetId:z.string(),clientRequestId:intentSchema}).parse(req.body),graph=readDocument(store,req.params.id,body.graphAssetId);if(graph.kind!=='graph')throw bad('Choose a production canvas.');const id=createHash('sha256').update(req.params.id+':'+body.clientRequestId).digest('hex'),old=store.getScopedRecord<GraphRun>('production-run',id);if(old&&old.data.graphAssetId!==body.graphAssetId)throw bad('This run identity belongs to a different canvas revision.',409);const now=new Date().toISOString(),run=old?.data??{id,projectId:req.params.id,graphAssetId:body.graphAssetId,createdAt:now,updatedAt:now,nodes:graph.nodes.map(n=>({nodeId:n.id,state:'waiting' as const,assetIds:[]}))};saveRun(store,run);res.json(await advanceGraph(store,media,run));});
 api.post('/projects/:id/production/runs/:runId/continue',async(req,res)=>{const row=store.getScopedRecord<GraphRun>('production-run',req.params.runId);if(!row||row.projectId!==req.params.id)throw bad('Production run not found.',404);res.json(await advanceGraph(store,media,row.data));});
 api.post('/projects/:id/production/runs/:runId/attach',(req,res)=>{const {nodeId,jobId}=z.object({nodeId:z.string(),jobId:z.string()}).parse(req.body),row=store.getScopedRecord<GraphRun>('production-run',req.params.runId);if(!row||row.projectId!==req.params.id)throw bad('Production run not found.',404);const node=row.data.nodes.find(n=>n.nodeId===nodeId),job=store.getJob(jobId);if(!node?.request||!job||job.projectId!==req.params.id||job.idempotencyKey!==createHash('sha256').update(`ots:${req.params.id}:${node.request.clientRequestId}`).digest('hex'))throw bad('This job does not belong to the planned node request.');if(node.jobId&&node.jobId!==job.id)throw bad('This node is already linked to another execution.',409);node.jobId=job.id;node.takeId=job.takeId;node.state='waiting';res.json(saveRun(store,row.data));});
 api.post('/projects/:id/production/deliveries',async(req,res)=>{const body=z.object({spec:deliverySpecSchema,clientRequestId:intentSchema,dueAt:z.string().datetime().optional()}).parse(req.body);if(body.dueAt){res.status(201).json({schedule:createSchedule(store,req.params.id,body.spec,body.dueAt,createHash('sha256').update(req.params.id+':'+body.clientRequestId).digest('hex'))});return;}res.status(201).json(await prepareDelivery(store,media,req.params.id,body.spec,body.clientRequestId));});
 api.patch('/projects/:id/production/deliveries/:deliveryId',(req,res)=>{const {action}=z.object({action:z.enum(['cancel','resume'])}).parse(req.body),row=store.getScopedRecord<DeliverySchedule>('delivery',req.params.deliveryId);if(!row||row.projectId!==req.params.id)throw bad('Delivery not found.',404);if(!['scheduled','paused','failed','cancelled'].includes(row.data.state))throw bad('A running or completed delivery cannot be changed.',409);const data={...row.data,state:action==='cancel'?'cancelled' as const:'scheduled' as const,error:undefined,updatedAt:new Date().toISOString()};store.upsertScopedRecord({...row,data,updatedAt:data.updatedAt});res.json(data);});
 api.use(reviewOwnerRoutes(store,media));return api;
}
