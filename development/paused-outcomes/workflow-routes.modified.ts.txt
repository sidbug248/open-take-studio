import { Router } from 'express';
import { createHash, randomUUID } from 'node:crypto';
import AdmZip from 'adm-zip';
import { z } from 'zod';
import type { Store } from './store.ts';
import type { LocalFileStorage } from './media.ts';
import type { Asset, Job, Take } from '../shared/types.ts';
import type { WorkflowVersion, WorkflowExecution } from '../shared/workflow-types.ts';
import { makeWorkflowVersion, rejectWorkflowSecrets, workflowRunSchema, workflowCreateSchema, WorkflowError } from './workflow-schema.ts';
import type { WorkflowPort } from './workflow-port.ts';
import type { WorkflowWorker } from './workflow-worker.ts';
import { summarizeInlineMedia } from './project-routes.ts';
import { NATIVE_OUTCOME_TEMPLATES } from './outcome-adapters/native-templates.ts';
import { installNativeOutcomeTemplate } from './outcome-adapters/native-install.ts';
import { validateNativeOutcomeInputs } from './outcome-adapters/native-validation.ts';

export function workflowFromAsset(asset:Asset):WorkflowVersion {
  if(asset.metadata.operation!=='comfy-workflow-version'||!asset.metadata.workflowVersion)throw new WorkflowError('This asset is not a native Comfy workflow version');
  const version=asset.metadata.workflowVersion as unknown as WorkflowVersion,reference=asset.metadata.cloudReference as {cloudEditorUrl?:string;cloudWorkspace?:{id:string;name:string}}|undefined;
  return reference?{...version,cloudEditorUrl:reference.cloudEditorUrl,cloudWorkspace:reference.cloudWorkspace}:version;
}
function workflowAsset(store:Store,id:string){const asset=store.getAsset(id);if(!asset)throw new WorkflowError('Workflow version not found',404);workflowFromAsset(asset);return asset;}
function requireNativeJob(store:Store,id:string){const job=store.getJob(id);if(!job||job.executionRoute!=='comfy-workflow'||!job.workflow)throw new WorkflowError('Native workflow job not found',404);return job;}
const publicJob=(store:Store,job:Job)=>summarizeInlineMedia({...job,archived:store.isArchivedJob(job.id)});
function creationBody(raw:any){if(raw?.format==='open-take-comfy-workflow')return Object.fromEntries(['name','description','editorGraph','apiGraph','bindings','outputNodeIds','outputKind','parentVersionAssetId','cloudEditorUrl','cloudWorkspace'].filter(key=>raw[key]!==undefined).map(key=>[key,raw[key]]));return raw;}

export function createWorkflowRouter(store:Store,media:LocalFileStorage,worker:WorkflowWorker,port:WorkflowPort):Router {
  const router=Router();
  router.get('/outcome-workflows',(_req,res)=>res.json(NATIVE_OUTCOME_TEMPLATES));
  router.post('/projects/:id/outcome-workflows/:templateId',async(req,res)=>res.status(201).json(await installNativeOutcomeTemplate(store,media,String(req.params.id),String(req.params.templateId))));
  router.get('/projects/:id/workflows',(req,res)=>{const projectId=String(req.params.id);if(!store.getProject(projectId))throw new WorkflowError('Project not found',404);res.json(store.listAssets(projectId).filter(asset=>asset.metadata.operation==='comfy-workflow-version').map(asset=>({asset,workflow:workflowFromAsset(asset)})));});
  router.post('/projects/:id/workflows',async(req,res)=>{
    const projectId=String(req.params.id),project=store.getProject(projectId);if(!project)throw new WorkflowError('Project not found',404);
    const workflow=makeWorkflowVersion(creationBody(req.body));let parentTakeId:string|undefined;
    if(workflow.parentVersionAssetId){const parent=workflowAsset(store,workflow.parentVersionAssetId);if(parent.projectId!==projectId)throw new WorkflowError('Parent workflow version belongs to another project');parentTakeId=store.listTakes(projectId).find(take=>take.assetIds.includes(parent.id))?.id;}
    const asset=await media.put({projectId,name:`${workflow.name}.workflow.json`,mimeType:'application/json',bytes:Buffer.from(JSON.stringify(workflow,null,2)),provenance:'local-edit',metadata:{operation:'comfy-workflow-version',workflowVersion:workflow,graphSha256:workflow.graphSha256}});
    const timestamp=new Date().toISOString();const take:Take={id:randomUUID(),projectId,name:`${workflow.name} · workflow version`,kind:'document',executionRoute:'comfy-workflow',workflowVersionAssetId:asset.id,parentTakeId,sourceAssetId:workflow.parentVersionAssetId,inputAssetIds:workflow.parentVersionAssetId?{previous_version:workflow.parentVersionAssetId}:{},assetIds:[asset.id],modelId:'local/comfy-workflow-definition',provider:'ComfyUI',prompt:workflow.description,settings:{graphSha256:workflow.graphSha256},status:'LOCAL',provenance:'local-edit',favorite:false,createdAt:timestamp,updatedAt:timestamp,changes:workflow.parentVersionAssetId?[{field:'workflowVersionAssetId',before:workflow.parentVersionAssetId,after:asset.id}]:[]};
    store.transaction(()=>{store.upsertAsset(asset);store.upsertTake(take);store.upsertProject({...store.getProject(projectId)!,updatedAt:timestamp});});res.status(201).json({asset,take,workflow});
  });
  router.get('/assets/:id/workflow',(req,res)=>{const asset=workflowAsset(store,String(req.params.id));res.json({asset,workflow:workflowFromAsset(asset)});});
  router.patch('/workflows/:id/cloud-reference',(req,res)=>{const asset=workflowAsset(store,String(req.params.id)),body=z.object({cloudEditorUrl:workflowCreateSchema.shape.cloudEditorUrl.unwrap(),cloudWorkspace:workflowCreateSchema.shape.cloudWorkspace}).strict().parse(req.body);const next={...asset,metadata:{...asset.metadata,cloudReference:{...body,recordedAt:new Date().toISOString(),association:'Explicit saved Cloud editor reference; execution evidence remains in the job record'}}};store.upsertAsset(next);res.json({asset:next,workflow:workflowFromAsset(next)});});
  router.get('/assets/:id/workflow/export',(req,res)=>{
    const asset=workflowAsset(store,String(req.params.id)),workflow=workflowFromAsset(asset),name=workflow.name.replace(/[^A-Za-z0-9_-]/g,'-').slice(0,100),format=String(req.query.format??'bundle');
    if(format==='editor'||format==='api'){res.attachment(`${name}.${format}.json`).type('application/json').send(JSON.stringify(format==='editor'?workflow.editorGraph:workflow.apiGraph,null,2));return;}
    if(format!=='bundle')throw new WorkflowError('Choose editor, api or bundle export');
    const zip=new AdmZip();zip.addFile('workflow.editor.json',Buffer.from(JSON.stringify(workflow.editorGraph,null,2)));zip.addFile('workflow.api.json',Buffer.from(JSON.stringify(workflow.apiGraph,null,2)));zip.addFile('workflow.studio.json',Buffer.from(JSON.stringify(workflow,null,2)));zip.addFile('dependencies.json',Buffer.from(JSON.stringify({target:workflow.target,dependencies:workflow.dependencies,graphSha256:workflow.graphSha256},null,2)));zip.addFile('README.txt',Buffer.from('Open Take Studio native Comfy workflow\nOpen workflow.editor.json in Comfy Cloud to inspect and edit the canvas. workflow.api.json is the execution graph. workflow.studio.json retains input bindings and dependency provenance. This version has not necessarily run; inspect its linked job records. API credentials are supplied only by the server. Models and services have their own terms.\n'));res.attachment(`${name}.comfy.zip`).type('application/zip').send(zip.toBuffer());
  });
  router.post('/workflows/:id/validate',async(req,res)=>{const workflow=workflowFromAsset(workflowAsset(store,String(req.params.id))),graph=structuredClone(workflow.apiGraph);for(const binding of workflow.bindings)if(['image','video','audio'].includes(binding.kind))graph[binding.nodeId].inputs[binding.inputName]={__type:'core/ASSET',info:{id:'input-required'}};res.json(await port.validate(graph));});
  router.post('/workflows/:id/run',async(req,res)=>{
    rejectWorkflowSecrets(req.body);const body=workflowRunSchema.parse(req.body),asset=workflowAsset(store,String(req.params.id)),workflow=workflowFromAsset(asset),projectId=asset.projectId;
    for(const id of [body.parentTakeId])if(id&&store.getTake(id)?.projectId!==projectId)throw new WorkflowError('Parent take belongs to another project');
    if(body.sourceAssetId&&store.getAsset(body.sourceAssetId)?.projectId!==projectId)throw new WorkflowError('Source asset belongs to another project');
    if(body.shotId&&store.getShot(body.shotId)?.projectId!==projectId)throw new WorkflowError('Shot belongs to another project');
    if(Object.keys(body.values).some(key=>!workflow.bindings.some(binding=>binding.key===key)))throw new WorkflowError('A supplied input is not exposed by this workflow version');
    const values:WorkflowExecution['values']={},inputAssetIds:Take['inputAssetIds']={workflow_version:asset.id};
    for(const binding of workflow.bindings){const value=body.values[binding.key]??binding.defaultValue;if(value===undefined){if(binding.required)throw new WorkflowError(`Choose ${binding.label}`);continue;}
      if(['image','video','audio'].includes(binding.kind)){if(typeof value!=='object')throw new WorkflowError(`${binding.label} needs a saved media asset`);const source=store.getAsset(value.assetId);if(!source||source.projectId!==projectId||source.kind!==binding.kind)throw new WorkflowError(`${binding.label} needs a ${binding.kind} from this project`);inputAssetIds[binding.key]=source.id;}
      else if(typeof value!==binding.kind)throw new WorkflowError(`${binding.label} needs a ${binding.kind}`);
      if(typeof value==='number'&&(binding.min!==undefined&&value<binding.min||binding.max!==undefined&&value>binding.max))throw new WorkflowError(`${binding.label} is outside the supported range`);
      if(binding.options&&typeof value==='string'&&!binding.options.includes(value))throw new WorkflowError(`${binding.label} is not a supported option`);values[binding.key]=value;
    }
    const key=`native:${projectId}:${body.clientRequestId}`,fingerprint=createHash('sha256').update(JSON.stringify({assetId:asset.id,...body,values})).digest('hex'),previous=store.getJobByIdempotencyKey(key);
    if(previous){if(previous.fingerprint!==fingerprint)throw new WorkflowError('This submission identity already belongs to different workflow inputs',409);res.json({job:publicJob(store,previous),take:store.getTake(previous.takeId)});return;}
    await validateNativeOutcomeInputs(workflow,values,store,media);
    // Recheck after asynchronous media validation to preserve one intended admission.
    const raced=store.getJobByIdempotencyKey(key);if(raced){if(raced.fingerprint!==fingerprint)throw new WorkflowError('This submission identity already belongs to different workflow inputs',409);res.json({job:publicJob(store,raced),take:store.getTake(raced.takeId)});return;}
    const timestamp=new Date().toISOString(),jobId=randomUUID(),takeId=randomUUID();
    const prompt=typeof values.prompt==='string'?values.prompt:workflow.description;
    const take:Take={id:takeId,projectId,name:body.name??workflow.name,kind:workflow.outputKind,executionRoute:'comfy-workflow',workflowVersionAssetId:asset.id,parentTakeId:body.parentTakeId,sourceAssetId:body.sourceAssetId,shotId:body.shotId,inputAssetIds,assetIds:[],modelId:`comfy-workflow/${workflow.graphSha256.slice(0,16)}`,provider:'Comfy Cloud',prompt,settings:{workflowVersionAssetId:asset.id,values},status:'PENDING',provenance:'generated',favorite:false,createdAt:timestamp,updatedAt:timestamp,jobId,changes:[]};
    if(body.parentTakeId){const parent=store.getTake(body.parentTakeId)!;if(parent.prompt!==prompt)take.changes.push({field:'prompt',before:parent.prompt,after:prompt});if(parent.workflowVersionAssetId!==asset.id)take.changes.push({field:'workflowVersionAssetId',before:parent.workflowVersionAssetId??null,after:asset.id});take.changes.push({field:'settings',before:parent.settings,after:take.settings});}
    const job:Job={id:jobId,projectId,takeId,executionRoute:'comfy-workflow',modelId:take.modelId,provider:'Comfy Cloud',idempotencyKey:key,fingerprint,inputs:{graph:workflow.apiGraph},state:'PENDING',attempts:0,createdAt:timestamp,updatedAt:timestamp,workflow:{versionAssetId:asset.id,graphSha256:workflow.graphSha256,target:'comfy-cloud',values,submissionOutcome:'not-submitted'}};
    store.saveTakeAndJob(take,job);res.status(202).json({take,job:publicJob(store,job)});void worker.tick();
  });
  router.get('/workflow-jobs/:id',(req,res)=>res.json(publicJob(store,requireNativeJob(store,String(req.params.id)))));
  router.post('/workflows/:id/import-execution',async(req,res)=>{
    const body=z.object({requestId:z.string().uuid(),name:z.string().max(180).optional(),inputAssetIds:z.record(z.string(),z.string().min(1).max(128)).default({}),parentTakeId:z.string().min(1).max(128).optional(),sourceAssetId:z.string().min(1).max(128).optional()}).strict().parse(req.body),asset=workflowAsset(store,String(req.params.id)),workflow=workflowFromAsset(asset),key=`native-import:${asset.projectId}:${body.requestId}`,previous=store.getJobByIdempotencyKey(key);
    if(previous){if(previous.workflow?.versionAssetId!==asset.id)throw new WorkflowError('This Cloud execution is already associated with a different local version',409);res.json({job:publicJob(store,previous),take:store.getTake(previous.takeId)});return;}
    const snapshot=await port.get(body.requestId);if(snapshot.status!=='succeeded'||snapshot.error)throw new WorkflowError('Only a successful existing Cloud execution can be imported');
    const executed=await port.executedGraph(body.requestId) as {format?:string;graph?:Record<string,any>;workflow?:Record<string,any>};rejectWorkflowSecrets(executed);const executedGraph=executed.graph??executed.workflow;
    const normalize=(graph:Record<string,any>)=>Object.fromEntries(Object.entries(graph).map(([id,node])=>[id,{class_type:node.class_type,inputs:node.inputs}]));const stable=(value:unknown):string=>JSON.stringify(value,(_,v)=>v&&typeof v==='object'&&!Array.isArray(v)?Object.fromEntries(Object.entries(v).sort(([a],[b])=>a.localeCompare(b))):v);
    if(executed.format!=='api'||!executedGraph||stable(normalize(executedGraph))!==stable(normalize(workflow.apiGraph)))throw new WorkflowError('Import the exact executed API graph as a workflow version before attaching this Cloud editor execution',409);
    const values:WorkflowExecution['values']={};
    if(body.parentTakeId&&store.getTake(body.parentTakeId)?.projectId!==asset.projectId)throw new WorkflowError('Imported source take belongs to another project');
    if(body.sourceAssetId&&store.getAsset(body.sourceAssetId)?.projectId!==asset.projectId)throw new WorkflowError('Imported source asset belongs to another project');
    for(const binding of workflow.bindings){const value=workflow.apiGraph[binding.nodeId].inputs[binding.inputName];if(['string','number','boolean'].includes(typeof value)&&['string','number','boolean'].includes(binding.kind))values[binding.key]=value as string|number|boolean;}
    for(const[key,aid]of Object.entries(body.inputAssetIds)){const binding=workflow.bindings.find(value=>value.key===key),source=store.getAsset(aid);if(!binding||!source||source.projectId!==asset.projectId||source.kind!==binding.kind)throw new WorkflowError('Imported workflow input does not match a media binding');const filename=workflow.apiGraph[binding.nodeId].inputs[binding.inputName];if(!port.hashInput||typeof filename!=='string')throw new WorkflowError('The imported source needs a verifiable Cloud content-addressed filename');const hash=(await port.hashInput(media.path(source))).replace(/^blake3:/,'');if(!/^[a-f0-9]{64}$/.test(hash)||!filename.split(/[\/]/).at(-1)?.startsWith(hash+'.'))throw new WorkflowError('The imported Cloud input filename does not match these source media bytes');values[key]={assetId:aid};}
    // Recheck after network I/O so concurrent imports share one local record.
    const admitted=store.getJobByIdempotencyKey(key);if(admitted){res.json({job:publicJob(store,admitted),take:store.getTake(admitted.takeId)});return;}
    const timestamp=new Date().toISOString(),jobId=randomUUID(),takeId=randomUUID();const take:Take={id:takeId,projectId:asset.projectId,name:body.name??`${workflow.name} · Cloud editor result`,kind:workflow.outputKind,executionRoute:'comfy-workflow',workflowVersionAssetId:asset.id,parentTakeId:body.parentTakeId,sourceAssetId:body.sourceAssetId,inputAssetIds:{workflow_version:asset.id,...body.inputAssetIds},assetIds:[],modelId:`comfy-workflow/${workflow.graphSha256.slice(0,16)}`,provider:'Comfy Cloud',prompt:typeof values.prompt==='string'?values.prompt:workflow.description,settings:{workflowVersionAssetId:asset.id,origin:'cloud-editor',values},status:'COLLECTING',provenance:'imported',favorite:false,createdAt:timestamp,updatedAt:timestamp,jobId,changes:[]};
    const job:Job={id:jobId,projectId:asset.projectId,takeId,executionRoute:'comfy-workflow',modelId:take.modelId,provider:'Comfy Cloud',requestId:snapshot.id,idempotencyKey:key,inputs:{graph:workflow.apiGraph},state:'COLLECTING',attempts:0,createdAt:timestamp,updatedAt:timestamp,result:snapshot,workflow:{origin:'cloud-editor',versionAssetId:asset.id,graphSha256:workflow.graphSha256,target:'comfy-cloud',values,cloudStatus:snapshot.status,submissionOutcome:'accepted'}};
    store.saveTakeAndJob(take,job);res.status(202).json({take,job:publicJob(store,job)});void worker.tick();
  });
  router.post('/workflow-jobs/:id/retry',(req,res)=>{const job=requireNativeJob(store,String(req.params.id));if(store.isArchivedJob(job.id))throw new WorkflowError('Imported workflow execution history is archived',409);if(worker.isActive(job.id))throw new WorkflowError('This native job is already active',409);if(job.workflow!.submissionOutcome==='unknown')throw new WorkflowError('An ambiguous submission requires its existing Cloud execution ID; retry cannot create another job',409);if(job.requestId&&['FAILED','EXPIRED','COMPLETED'].includes(job.state))throw new WorkflowError('This execution has finished. Branch a new take to intentionally run again.',409);const next=store.patchJob(job.id,{state:job.requestId?'IN_PROGRESS':'PENDING',nextPollAt:undefined,errorType:undefined,errorMessage:undefined});res.json(publicJob(store,next));void worker.tick();});
  router.post('/workflow-jobs/:id/attach',async(req,res)=>{const job=requireNativeJob(store,String(req.params.id));if(store.isArchivedJob(job.id)||job.requestId||job.workflow!.submissionOutcome!=='unknown')throw new WorkflowError('Only an unresolved native submission can attach an existing execution',409);const requestId=z.object({requestId:z.string().uuid()}).parse(req.body).requestId;const snapshot=await port.get(requestId),executed=await port.executedGraph(requestId) as {format?:string;graph?:unknown;workflow?:unknown};const expected=job.inputs.graph;const stable=(value:unknown):string=>JSON.stringify(value,(_,v)=>v&&typeof v==='object'&&!Array.isArray(v)?Object.fromEntries(Object.entries(v).sort(([a],[b])=>a.localeCompare(b))):v);if(executed.format!=='api'||stable(executed.graph??executed.workflow)!==stable(expected))throw new WorkflowError('The existing Cloud execution graph does not match this intended submission',409);const next=store.patchJob(job.id,{requestId:snapshot.id,state:'IN_PROGRESS',nextPollAt:undefined,errorType:undefined,errorMessage:undefined,workflow:{...job.workflow!,submissionOutcome:'accepted'}});res.json(publicJob(store,next));void worker.tick();});
  router.post('/workflow-jobs/:id/cancel',async(req,res)=>{const job=requireNativeJob(store,String(req.params.id));if(store.isArchivedJob(job.id))throw new WorkflowError('Imported execution history is archived',409);if(!job.requestId){if(worker.isActive(job.id)||job.workflow!.submissionOutcome==='unknown')throw new WorkflowError('Submission is unresolved; locate the Cloud execution before canceling',409);const next=store.patchJob(job.id,{state:'FAILED',errorType:'canceled',errorMessage:'Canceled before Cloud submission.'});res.json(publicJob(store,next));return;}const result=await port.cancel(job.requestId);const next=store.patchJob(job.id,{state:'IN_PROGRESS',nextPollAt:undefined,workflow:{...job.workflow!,cloudStatus:result.status}});res.json(publicJob(store,next));void worker.tick();});
  return router;
}
