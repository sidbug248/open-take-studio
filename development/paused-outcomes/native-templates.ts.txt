function editableGraph(api:WorkflowGraph){
  const links:any[]=[];let linkId=0;
  const nodes=Object.entries(api).map(([id,node],index)=>{const info=snapshot.nodes[node.class_type],inputLinks:any[]=[];
    for(const[field,value]of Object.entries(node.inputs))if(Array.isArray(value)&&value.length===2){const producer=snapshot.nodes[api[String(value[0])]!.class_type],type=producer.output[value[1]];const lid=++linkId;links.push([lid,Number(value[0]),value[1],Number(id),inputLinks.length,type]);inputLinks.push({name:field,type,link:lid});}
    const widgets=Object.fromEntries(Object.entries(node.inputs).filter(([,v])=>!Array.isArray(v)));
    return{id:Number(id),type:node.class_type,title:node._meta?.title??info.display_name,pos:[index*500,90],size:[440,480],flags:{},order:index,mode:0,inputs:inputLinks,outputs:(info.output??[]).map((type:string,slot:number)=>({name:info.output_name?.[slot]??type,type,links:[],slot_index:slot})),properties:{'Node name for S&R':node.class_type},widgets_values:Object.values(widgets),widgets_values_named:widgets};
  });
  for(const link of links){const n=nodes.find(n=>n.id===link[1])!;n.outputs[link[2]].links.push(link[0]);}
  return{id:createHash('sha256').update(JSON.stringify(api)).digest('hex').slice(0,32),revision:0,last_node_id:Math.max(...nodes.map(n=>n.id)),last_link_id:linkId,nodes,links,groups:[],config:{},extra:{ds:{scale:.7,offset:[30,50]},openTakeStudio:{source:'verified-native-outcome-template',schemaVerifiedAt:snapshot.verifiedAt}},version:.4};
}
